// ═════════════════════════════════════════════════════════════════════════════
//  PREISE / BASIS  (unverändert zu v6)
// ═════════════════════════════════════════════════════════════════════════════
realTk = ticker.standard(syminfo.tickerid)
[rO, rH, rL, rC] = request.security(realTk, timeframe.period, [open, high, low, close])
h200   = request.security(realTk, "60", ta.ema(close, 200)[1], lookahead = barmerge.lookahead_on)
utcH   = hour(time, "UTC")
utcMin = utcH * 60 + minute(time, "UTC")
trR    = math.max(rH - rL, math.abs(rH - nz(rC[1], rC)), math.abs(rL - nz(rC[1], rC)))
atrV   = ta.rma(trR, 14)
upT    = rC > h200
dnT    = rC < h200
ok     = barstate.isconfirmed

// ── 1) FVG-Retest: Lücke (Kerze i-2 ↔ i), Rücktest innerhalb 12 Bars, Schluss über/unter der Lücke
var float fbTop = na
var float fbBot = na
var float fsTop = na
var float fsBot = na
var int   fbBar = -100000
var int   fsBar = -100000
fvgB = bar_index - fbBar >= 1 and bar_index - fbBar <= 13 and rL <= fbTop and rL >= fbBot and rC > rO and rC > fbTop
fvgS = bar_index - fsBar >= 1 and bar_index - fsBar <= 13 and rH >= fsBot and rH <= fsTop and rC < rO and rC < fsBot
bool newBullFvg = rL > rH[2]
bool newBearFvg = rH < rL[2]
if newBullFvg
    fbTop := rL
    fbBot := rH[2]
    fbBar := bar_index
if newBearFvg
    fsBot := rH
    fsTop := rL[2]
    fsBar := bar_index
s1B = fvgB and upT
s1S = fvgS and not fvgB and dnT

// ── 2) EMA-Cross 9/21 über/unter EMA200 (5m), 07–20 UTC
e9   = ta.ema(rC, 9)
e21  = ta.ema(rC, 21)
e200 = ta.ema(rC, 200)
crU  = e9 > e21 and e9[1] <= e21[1]
crD  = e9 < e21 and e9[1] >= e21[1]
s2B  = crU and rC > e200 and utcH >= 7 and utcH < 20 and upT
s2S  = crD and rC < e200 and utcH >= 7 and utcH < 20 and dnT

// ── 3) Bollinger-Squeeze: Bandbreite im unteren 20-%-Bereich (500 Bars), Schluss außerhalb des Bands
bbM  = ta.sma(rC, 20)
bbD  = ta.stdev(rC, 20, false)
bbW  = 4 * bbD / bbM
bbQ  = ta.percentile_linear_interpolation(bbW, 500, 20)
sq   = bbW[1] <= bbQ[1]
s3B  = sq and rC > bbM + 2 * bbD and upT
s3S  = sq and rC < bbM - 2 * bbD and not s3B and dnT

// ── 4) Trend-Pullback NY: Impuls > 2 ATR in 6 Bars, Rücksetzer an EMA20, Schluss in Trendrichtung, 12–17 UTC
e20  = ta.ema(rC, 20)
impU = ta.highest(rC - rC[6] > 2 * atrV ? 1 : 0, 6) > 0
impD = ta.highest(rC[6] - rC > 2 * atrV ? 1 : 0, 6) > 0
pbB  = impU and rL <= e20 and rC > e20 and rC > rO
pbS  = impD and rH >= e20 and rC < e20 and rC < rO and not pbB
nyS  = utcH >= 12 and utcH < 17
s4B  = pbB and nyS and upT
s4S  = pbS and nyS and dnT

// ── 5) RSI NY spät: Heikin-Ashi-RSI 5 kommt aus 30/70 zurück, HA-Farbe bestätigt, 17–21 UTC, ATR ≥ 1.5 $
haC  = (rO + rH + rL + rC) / 4
float haO = na
haO := na(haO[1]) ? (rO + rC) / 2 : (haO[1] + haC[1]) / 2
rsi5 = ta.rsi(haC, 5)
lt   = utcH >= 17 and utcH < 21 and atrV >= 1.5
r5B  = rsi5[1] <= 30 and rsi5 > 30 and haC > haO and lt and upT
r5S  = rsi5[1] >= 70 and rsi5 < 70 and haC < haO and lt and dnT and not r5B

sigB = array.from(on1 and ok and s1B, on2 and ok and s2B, on3 and ok and s3B, on4 and ok and s4B, on5 and ok and r5B)
sigS = array.from(on1 and ok and s1S, on2 and ok and s2S, on3 and ok and s3S, on4 and ok and s4S, on5 and ok and r5S)
SLP  = array.from(sl1, sl2, sl3, sl4, sl5)
RTS  = array.from(rt1, rt2, rt3, rt4, rt5)
MULT = array.from(1.0, 1.5, 2.0, 3.0)

// ── 5m-Freigabe für NEUE Signale (Ausstiege laufender Trades sind nie blockiert)
bool tfOk      = not only5m or timeframe.period == "5"
bool rollBlock = useRoll and utcMin >= 20 * 60 + 45 and utcMin < 23 * 60 + 15
bool spikeBlk  = useSpike and not na(atrV[1]) and (rH - rL) > spikeK * atrV[1]
bool friBlock  = useFri and dayofweek(time, "UTC") == dayofweek.friday and utcH >= 20
bool allowNew  = tfOk and not rollBlock and not spikeBlk and not friBlock

// ═════════════════════════════════════════════════════════════════════════════
//  TRADE-ENGINE – je Setup ein eigener Trade (4 × 25 % an TP1–TP4)  (Logik wie v6)
// ═════════════════════════════════════════════════════════════════════════════
var array<int>   sDir   = array.new<int>(NS, 0)
var array<int>   sBar   = array.new<int>(NS, 0)
var array<int>   sStage = array.new<int>(NS, 0)
var array<int>   sLast  = array.new<int>(NS, -100000)
var array<float> sSig   = array.new<float>(NS, na)
var array<float> sFill  = array.new<float>(NS, na)
var array<float> sSL    = array.new<float>(NS, na)
var array<float> sBE    = array.new<float>(NS, na)
var array<float> sStop  = array.new<float>(NS, na)
var array<float> sTP    = array.new<float>(NS * 4, na)
var array<int>   stN    = array.new<int>(NS, 0)
var array<int>   stW    = array.new<int>(NS, 0)
var array<float> stR    = array.new<float>(NS, 0.0)
var array<int>   sTm    = array.new<int>(NS, 0)      // Einstiegszeit (nur Anzeige)

var int dayCnt = 0
var int nDays  = 1
if dayofmonth(time, "UTC") != dayofmonth(time[1], "UTC")
    dayCnt := 0
    nDays  += 1

f_side(int d) =>
    d == 1 ? "BUY" : "SELL"
f_p(float x) =>
    str.tostring(x, format.mintick)
f_r(float x) =>
    (x >= 0 ? "+" : "") + str.tostring(x, "0.00") + "R"
// JSON-sicheres Secret (für normale Secrets identisch zu v6)
secretJs = str.replace_all(str.replace_all(whSecret, "\\", "\\\\"), "\"", "\\\"")
// Nachricht für Webhook: Levels = Entry ± Abstand (wie beim Broker eingegeben) – Format unverändert zu v6
f_js(string ev, int k, int d, float px) =>
    slD = SLP.get(k) * pipSz
    tp1 = slD * RTS.get(k)
    e   = sSig.get(k)
    '{"secret":"' + secretJs + '","event":"' + ev + '","setup":"' + NAMES.get(k) + '","side":"' + f_side(d) + '","symbol":"' + syminfo.ticker +
     '","price":' + f_p(px) + ',"entry":' + f_p(e) + ',"sl":' + f_p(e - d * slD) +
     ',"tp1":' + f_p(e + d * tp1) + ',"tp2":' + f_p(e + d * tp1 * 1.5) + ',"tp3":' + f_p(e + d * tp1 * 2) + ',"tp4":' + f_p(e + d * tp1 * 3) + '}'
f_hit(float y, string txt, color bg, string tip) =>
    // auf der offenen Echtzeitkerze vorläufig (kann bis Kerzenschluss noch kippen)
    if showHits
        label.new(bar_index, y, ok ? txt : txt + " …", style = label.style_label_center, color = color.new(bg, ok ? 10 : 55), textcolor = color.white, size = size.tiny, tooltip = ok ? tip : tip + "\n(vorläufig – Kerze noch offen)")
// Broker-Level (wie im Webhook, ohne Spread): Entry = Signal-Schluss
f_brkSL(int k, int d) =>
    sSig.get(k) - d * SLP.get(k) * pipSz
f_brkTP(int k, int d, int j) =>
    sSig.get(k) + d * SLP.get(k) * pipSz * RTS.get(k) * MULT.get(j)

// ── Ausstiege
for k = 0 to NS - 1
    d = sDir.get(k)
    if d != 0 and bar_index > sBar.get(k)
        bool chk = false
        slD0 = SLP.get(k) * pipSz
        if sStage.get(k) == 0
            sl  = sSL.get(k)
            t1  = sTP.get(k * 4)
            if d == 1 ? rL <= sl : rH >= sl
                // realistischer Ausstieg: Eröffnung bereits jenseits des SL (Gap) -> Ausstieg ab Eröffnungskurs
                bool gapped = gapReal and (d == 1 ? rO <= sl : rO >= sl)
                float lossR = gapped ? (d == 1 ? rO - sFill.get(k) : sFill.get(k) - (rO + spread)) / slD0 : -1.0
                stN.set(k, stN.get(k) + 1)
                stR.set(k, stR.get(k) + lossR)
                sDir.set(k, 0)
                sLast.set(k, bar_index)
                f_hit(f_brkSL(k, d), "✕ SL", cSell, NAMES.get(k) + " " + f_side(d) + " · SL " + f_p(f_brkSL(k, d)) + " · " + f_r(lossR) + (gapped ? " (Gap)" : ""))
                alert(f_js("SL", k, d, sSig.get(k) - d * SLP.get(k) * pipSz), alert.freq_once_per_bar_close)
            else if d == 1 ? rH >= t1 : rL <= t1
                chk := true
                sLast.set(k, bar_index)
        else
            stp = sStop.get(k)
            if d == 1 ? rL <= stp : rH >= stp
                // Stopp wird erst zum Schluss der TP1-Kerze nachgezogen (wie per Webhook): liegt die Eröffnung
                // schon jenseits, erfolgt der Ausstieg zum Eröffnungskurs (gilt für BE- und Original-SL)
                bool  gapped = gapReal and (d == 1 ? rO <= stp : rO >= stp)
                float rest   = 1 - 0.25 * sStage.get(k)
                if gapped
                    stR.set(k, stR.get(k) + rest * (d == 1 ? rO - sFill.get(k) : sFill.get(k) - (rO + spread)) / slD0)
                else if not useBE
                    stR.set(k, stR.get(k) - rest)
                sDir.set(k, 0)
                // nach TP1 zählt der Trade als Gewinner – Rückkehr zum Einstieg ist "BE HIT", kein SL
                float shownStop = useBE ? sSig.get(k) : f_brkSL(k, d)
                f_hit(shownStop, useBE ? "BE ✓" : "✕ SL", useBE ? #546E7A : cSell, NAMES.get(k) + " " + f_side(d) + " · Rest " + str.tostring(rest * 100, "0") + " % bei " + f_p(shownStop) + (gapped ? " (Ausstieg ab Eröffnung " + f_p(rO) + ")" : ""))
                alert(f_js(useBE ? "BE" : "SL", k, d, sSig.get(k)), alert.freq_once_per_bar_close)
            else
                chk := true
        if chk
            s0 = sStage.get(k)
            for j = s0 to 3
                lv = sTP.get(k * 4 + j)
                if not (d == 1 ? rH >= lv : rL <= lv)
                    break
                sStage.set(k, j + 1)
                if j == 0
                    stN.set(k, stN.get(k) + 1)
                    stW.set(k, stW.get(k) + 1)
                stR.set(k, stR.get(k) + 0.25 * RTS.get(k) * MULT.get(j))
                if hitMode == "Alle TPs" or j == 0 or j == 3
                    f_hit(f_brkTP(k, d, j), "TP" + str.tostring(j + 1) + " ✓", COLS.get(k), NAMES.get(k) + " " + f_side(d) + " · TP" + str.tostring(j + 1) + " " + f_p(f_brkTP(k, d, j)) + " · +" + str.tostring(RTS.get(k) * MULT.get(j), "0.0#") + "R auf 25 %")
                alert(f_js("TP" + str.tostring(j + 1), k, d, sSig.get(k) + d * SLP.get(k) * pipSz * RTS.get(k) * MULT.get(j)), alert.freq_once_per_bar_close)
            if sStage.get(k) == 4
                sDir.set(k, 0)
            else if s0 == 0
                sStop.set(k, useBE ? sBE.get(k) : sSL.get(k))

// ── Neue Signale (je Setup: kein Trade offen ODER laufender Trade hat TP1)
for k = 0 to NS - 1
    d  = sDir.get(k)
    st = sStage.get(k)
    b  = sigB.get(k)
    s  = sigS.get(k)
    if (b or s) and (d == 0 or st >= 1) and bar_index - sLast.get(k) > cooldown and dayCnt < maxDay and allowNew
        slD = SLP.get(k) * pipSz
        if d != 0
            ex = d == 1 ? rC : rC + spread
            stR.set(k, stR.get(k) + (1 - 0.25 * st) * (d == 1 ? ex - sFill.get(k) : sFill.get(k) - ex) / slD)
        nd   = b ? 1 : -1
        fill = nd == 1 ? rC + spread : rC
        sDir.set(k, nd)
        sBar.set(k, bar_index)
        sTm.set(k, time)
        sStage.set(k, 0)
        sSig.set(k, rC)
        sFill.set(k, fill)
        sSL.set(k, nd == 1 ? fill - slD : fill + slD - spread)
        sBE.set(k, nd == 1 ? fill : fill - spread)
        sStop.set(k, sSL.get(k))
        for j = 0 to 3
            tpd = slD * RTS.get(k) * MULT.get(j)
            sTP.set(k * 4 + j, nd == 1 ? fill + tpd : fill - tpd - spread)
        dayCnt += 1
        // Signal-Label: Setup-Farbe + Pfeil (Richtung auch ohne Farbe erkennbar), Details per Tooltip
